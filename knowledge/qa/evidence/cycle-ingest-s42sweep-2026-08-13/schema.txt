            raw_content     TEXT    NOT NULL,
            content_hash    TEXT    NOT NULL,
            tags            TEXT,
            ingested_at     TEXT    NOT NULL,
            UNIQUE(source_file, source_heading)

            ON lesson_entries(content_hash);
